       export function useLocale() {
         const { i18n } = useTranslation();
         const currentLocale = i18n.language as "en" | "bn";
         const switchLanguage = (locale: "en" | "bn") => i18n.changeLanguage(locale);
         const toggleLanguage = () => switchLanguage(currentLocale === "en" ? "bn" : "en");

    5. Create date formatting utility (`lib/dates.ts`):
       ```typescript
       import { format as fnsFormat, formatDistance, formatRelative } from "date-fns";
       import { bn } from "date-fns/locale";
       import { enUS } from "date-fns/locale";

       const localeMap = { en: enUS, bn };

       export function formatDate(date: Date | string, formatStr: string, locale: "en" | "bn"): string {
         const d = typeof date === "string" ? new Date(date) : date;
         return fnsFormat(d, formatStr, { locale: localeMap[locale] });
       }

       export function formatRelativeDate(date: Date | string, baseDate: Date, locale: "en" | "bn"): string {
         const d = typeof date === "string" ? new Date(date) : date;
         return formatRelative(d, baseDate, { locale: localeMap[locale] });
       }
       ```

    6. Create number formatting utility (`lib/format.ts`):
       ```typescript
       // Most Bangladeshi apps use Western digits even in Bengali UI
       // Use 'en-BD' for Western digits, 'bn-BD' for Bengali digits
       export function formatNumber(num: number, locale: "en" | "bn"): string {
         return new Intl.NumberFormat(locale === "bn" ? "en-BD" : "en-US").format(num);
       }

       export function formatCycleDay(day: number, locale: "en" | "bn"): string {
         return formatNumber(day, locale);
       }
       ```
