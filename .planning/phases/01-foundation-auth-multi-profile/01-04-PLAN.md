       ```typescript
       export async function checkProfileOwnership(req: Request, res: Response, next: NextFunction) {
         const { profileId } = req.params;
         const userId = req.user!.id; // Set by verifyJWT

         const profile = await prisma.profile.findFirst({
           where: { id: profileId, userId },
         });

         if (!profile) {
           return res.status(403).json({ success: false, error: "Profile access denied" });
         }

         req.profile = profile; // Attach for downstream handlers
         next();
       }
       ```

    3. Create profile-aware query hook (`hooks/useProfileQuery.ts`):
       ```typescript
       export function useProfileQuery<T>(
         key: string[],
         queryFn: (profileId: string) => Promise<T>,
         options?: Omit<UseQueryOptions<T>, "queryKey" | "queryFn">
       ) {
         const activeProfileId = useProfileStore((s) => s.activeProfileId);
         return useQuery({
           queryKey: [...key, activeProfileId],
           queryFn: () => queryFn(activeProfileId!),
           enabled: !!activeProfileId,
           ...options,
         });
       }
       ```
       This hook ensures ALL profile-scoped queries include activeProfileId in the key, so cache invalidation on profile switch works automatically.
